💻 C Program Used for Student Information

#include <stdio.h>

struct Student
{
    char name[50];
    char regNo[20];
    char address[100];
    char phone[20];
    int age;
    char course[50];
    float programming;
    float mathematics;
    float database;
    float networks;
    float attendance;
};

int main()
{
    struct Student s;

    printf("Enter Student Name: ");
    scanf(" %[^\n]", s.name);

    printf("Enter Register Number: ");
    scanf("%s", s.regNo);

    printf("Enter Address: ");
    scanf(" %[^\n]", s.address);

    printf("Enter Phone Number: ");
    scanf("%s", s.phone);

    printf("Enter Age: ");
    scanf("%d", &s.age);

    printf("Enter Course: ");
    scanf(" %[^\n]", s.course);

    printf("Enter Programming Mark: ");
    scanf("%f", &s.programming);

    printf("Enter Mathematics Mark: ");
    scanf("%f", &s.mathematics);

    printf("Enter Database Mark: ");
    scanf("%f", &s.database);

    printf("Enter Computer Networks Mark: ");
    scanf("%f", &s.networks);

    printf("Enter Attendance: ");
    scanf("%f", &s.attendance);

    printf("\n----- STUDENT INFORMATION -----\n");

    printf("Name: %s\n", s.name);
    printf("Register No: %s\n", s.regNo);
    printf("Address: %s\n", s.address);
    printf("Phone: %s\n", s.phone);
    printf("Age: %d\n", s.age);
    printf("Course: %s\n", s.course);

    printf("\n----- MARKS -----\n");

    printf("Programming: %.2f\n", s.programming);
    printf("Mathematics: %.2f\n", s.mathematics);
    printf("Database: %.2f\n", s.database);
    printf("Networks: %.2f\n", s.networks);

    printf("\nAttendance: %.2f%%\n", s.attendance);

    return 0;
}
